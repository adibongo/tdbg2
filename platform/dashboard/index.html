period = (w, g) => { const s = wkStr(w); const y = s.slice(0, 4), m = +s.slice(5, 7);
  return g === "week" ? s : g === "month" ? s.slice(0, 7) : g === "quarter" ? `${y}-Q${Math.ceil(m / 3)}` : g === "year" ? y : "all" }
sqlText = {
  const sel = []; const grp = [];
  if (grain !== "(none)") { sel.push(`date_trunc('${grain}', week_starting) AS period`); grp.push("period") }
  if (dim !== "(none)") { const col = dim === "depot" ? "depot_name" : dim === "promo" ? "on_promotion" : dim; sel.push(`${col} AS ${dim}`); grp.push(dim) }
  sel.push(`${layer.metrics[met].sql} AS ${met}`)
  return `SELECT\n  ${sel.join(",\n  ")}\nFROM ${layer.table}` + (grp.length ? `\nGROUP BY ${grp.join(", ")}\nORDER BY ${grp[0]}` : "")
}

index = {
  const tok = s => s.toLowerCase().replace(/[^a-z0-9 ]/g, " ").split(/\s+/).filter(w => w.length > 2)
  const docs = reviews.map(r => tok(r.txt)); const df = new Map()
  for (const d of docs) for (const w of new Set(d)) df.set(w, (df.get(w) || 0) + 1)
  const idf = w => Math.log(reviews.length / (df.get(w) || reviews.length)) + 1
  const vec = d => { const v = new Map(); for (const w of d) if (df.has(w)) v.set(w, (v.get(w) || 0) + idf(w));
    const n = Math.hypot(...v.values()); for (const [k, x] of v) v.set(k, x / (n || 1)); return v }
  return {tok, vec, vecs: docs.map(vec)}
}
hits = {

}

agg = rows => {
  const s = {rev: 0, gp: 0, units: 0, n: 0, so: 0, disc: 0, rev_promo: 0, n_promo: 0}
  for (const r of rows) { s.rev += r.rev; s.gp += r.gp; s.units += r.units; s.n += r.n; s.so += r.so;
    if (r.promo === 1) { s.rev_promo += r.rev; s.n_promo += r.n; s.disc += (r.disc || 0) } }
  return s
}
METRIC = {
  // JS mirrors of the SQL in metrics.yml: same names, and sum first, divide last
  const m = {}
  m.revenue = s => s.rev
  m.units = s => s.units
  m.gross_profit = s => s.gp
  m.gross_margin_pct = s => 100 * s.gp / s.rev
  m.avg_selling_price = s => s.rev / s.units
  m.stockout_rate_pct = s => 100 * s.so / s.n
  m.promo_share_pct = s => 100 * s.rev_promo / s.rev
  m.avg_discount_pct = s => s.n_promo ? 100 * s.disc / s.n_promo : NaN
  return m
}
